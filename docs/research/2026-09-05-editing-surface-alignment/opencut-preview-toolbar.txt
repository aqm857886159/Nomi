    62	function TimecodeDisplay() {
    63		const editor = useEditor();
    64		const totalDuration = useEditor((e) => e.timeline.getTotalDuration());
    65		const fps = useEditor((e) => e.project.getActive().settings.fps);
    66		const [currentTime, setCurrentTime] = useState<MediaTime>(() =>
    67			editor.playback.getCurrentTime(),
    68		);
    69	
    70		useEffect(() => {
    71			const unsubscribeUpdate = editor.playback.onUpdate(setCurrentTime);
    72			const unsubscribeSeek = editor.playback.onSeek(setCurrentTime);
    73			return () => {
    74				unsubscribeUpdate();
    75				unsubscribeSeek();
    76			};
    77		}, [editor.playback]);
    78	
    79		return (
    80			<div className="flex items-center">
    81				<EditableTimecode
    82					time={currentTime}
    83					duration={totalDuration}
    84					format="HH:MM:SS:FF"
    85					fps={fps}
    86					onTimeChange={({ time }) => editor.playback.seek({ time })}
    87					className="text-center"
    88				/>
    89				<span className="text-muted-foreground px-2 font-mono text-xs">/</span>
    90				<span className="text-muted-foreground font-mono text-xs">
    91					{formatTimecode({
    92						time: totalDuration,
    93						format: "HH:MM:SS:FF",
    94						rate: fps,
    95					})}
    96				</span>
    97			</div>
    98		);
    99	}
   100	
   101	function ZoomSelect() {
   102		const { isAtFit, zoomPercent, fitToScreen, setViewportPercent } =
   103			usePreviewViewport();
   104	
   105		const displayLabel = isAtFit ? "Fit" : `${zoomPercent}%`;
   106	
   107		const onValueChange = (value: string) => {
   108			if (value === "fit") {
   109				fitToScreen();
   110			} else {
   111				setViewportPercent({ percent: Number(value) });
   112			}
   113		};
   114	
   115		return (
   116			<Select
   117				value={isAtFit ? "fit" : String(zoomPercent)}
   118				onValueChange={onValueChange}
   119			>
   120				<SelectTrigger className="tabular-nums">{displayLabel}</SelectTrigger>
   121				<SelectContent>
   122					<SelectItem value="fit">Fit</SelectItem>
   123					<SelectSeparator />
   124					{PREVIEW_ZOOM_PRESETS.map((preset) => (
   125						<SelectItem key={preset} value={String(preset)}>
   126							{preset}%
   127						</SelectItem>
   128					))}
   129				</SelectContent>
   130			</Select>
   131		);
   132	}
   133	
   134	function PlayPauseButton() {
   135		const isPlaying = useEditor((e) => e.playback.getIsPlaying());
   136	
   137		return (
   138			<Button
   139				variant="text"
   140				size="icon"
   141				onClick={() => invokeAction("toggle-play")}
   142			>
   143				<HugeiconsIcon icon={isPlaying ? PauseIcon : PlayIcon} />
   144			</Button>
   145		);
   146	}
