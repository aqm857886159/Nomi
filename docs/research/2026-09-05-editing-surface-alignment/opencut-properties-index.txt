     1	"use client";
     2	
     3	import { ScrollArea } from "@/components/ui/scroll-area";
     4	import { Button } from "@/components/ui/button";
     5	import {
     6		Tooltip,
     7		TooltipContent,
     8		TooltipProvider,
     9		TooltipTrigger,
    10	} from "@/components/ui/tooltip";
    11	import { useEditor } from "@/editor/use-editor";
    12	import { useElementSelection } from "@/timeline/hooks/element/use-element-selection";
    13	import { usePropertiesStore } from "./stores/properties-store";
    14	import { getPropertiesConfig } from "./registry";
    15	import { cn } from "@/utils/ui";
    16	import { EmptyView } from "./empty-view";
    17	
    18	export function PropertiesPanel() {
    19		const editor = useEditor();
    20		useEditor((e) => e.scenes.getActiveSceneOrNull());
    21		useEditor((e) => e.media.getAssets());
    22		const { selectedElements } = useElementSelection();
    23		const { activeTabPerType, setActiveTab } = usePropertiesStore();
    24	
    25		if (selectedElements.length === 0) {
    26			return (
    27				<div className="panel bg-background flex h-full flex-col items-center justify-center overflow-hidden rounded-sm border">
    28					<EmptyView />
    29				</div>
    30			);
    31		}
    32	
    33		if (selectedElements.length > 1) {
    34			return (
    35				<div className="panel bg-background flex h-full flex-col items-center justify-center overflow-hidden rounded-sm border">
    36					<p className="text-muted-foreground text-sm">
    37						{selectedElements.length} elements selected.0
    38					</p>
    39				</div>
    40			);
    41		}
    42	
    43		const mediaAssets = editor.media.getAssets();
    44	
    45		const elementsWithTracks = editor.timeline.getElementsWithTracks({
    46			elements: selectedElements,
    47		});
    48		const elementWithTrack = elementsWithTracks[0];
    49	
    50		if (!elementWithTrack) return null;
    51	
    52		const { element, track } = elementWithTrack;
    53		const config = getPropertiesConfig({ element, mediaAssets });
    54		const visibleTabs = config.tabs;
    55	
    56		const storedTabId = activeTabPerType[element.type];
    57		const isStoredTabVisible = visibleTabs.some((t) => t.id === storedTabId);
    58		const activeTabId = isStoredTabVisible ? storedTabId : config.defaultTab;
    59		const activeTab =
    60			visibleTabs.find((t) => t.id === activeTabId) ?? visibleTabs[0];
    61	
    62		if (!activeTab) return null;
    63	
    64		return (
    65			<div className="panel bg-background flex h-full overflow-hidden rounded-sm border">
    66				<TooltipProvider delayDuration={0}>
    67					<div className="flex shrink-0 flex-col gap-0.5 border-r p-1 scrollbar-hidden overflow-y-auto">
    68						{visibleTabs.map((tab) => (
    69							<Tooltip key={tab.id}>
    70								<TooltipTrigger asChild>
    71									<Button
    72										variant={tab.id === activeTab.id ? "secondary" : "ghost"}
    73										size="icon"
    74										onClick={() =>
    75											setActiveTab({
    76												elementType: element.type,
    77												tabId: tab.id,
    78											})
    79										}
    80										aria-label={tab.label}
    81										className={cn(
    82											"shrink-0",
    83											"h-8 w-8",
    84											tab.id !== activeTab.id && "text-muted-foreground",
    85										)}
    86									>
    87										{tab.icon}
    88									</Button>
    89								</TooltipTrigger>
    90								<TooltipContent side="right">{tab.label}</TooltipContent>
    91							</Tooltip>
    92						))}
    93					</div>
    94				</TooltipProvider>
    95				<ScrollArea className="flex-1 scrollbar-hidden">
    96					{activeTab.content({ trackId: track.id })}
    97				</ScrollArea>
    98			</div>
    99		);
   100	}
